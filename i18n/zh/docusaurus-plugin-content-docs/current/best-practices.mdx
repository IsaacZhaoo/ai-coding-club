### 1. 永远不要信任 AI 处理安全关键代码

**❌ 不要这样做：**
```python
# AI 可能生成：

    grant_access()
```

**✅ 应该这样做：**
```python
# 始终审查安全代码
from werkzeug.security import check_password_hash
import os

# 审查 AI 输出是否存在：
# - 硬编码的密钥
# - SQL 注入风险
# - XSS 漏洞
# - 不当的身份验证

def verify_password(username, password):
    """
    安全的密码验证
    AI 生成的代码已审查和增强
    """
    user = User.query.filter_by(username=username).first()

    if not user:
        # 防止用户名枚举
        check_password_hash("dummy", password)
        return False

    # 使用恒定时间比较
    return check_password_hash(user.password_hash, password)
```

## 🚫 常见陷阱和解决方案

### 陷阱 1：盲目复制代码

**问题：**
```typescript
// AI 生成了这个 React hook
function useData() {
  const [data, setData] = useState([]);

  useEffect(() => {
    fetch('/api/data').then(r => r.json()).then(setData);
  }, []); // 缺少错误处理、竞态条件

  return data;
}
```

**有什么问题：**
- ❌ 没有错误处理
- ❌ 没有加载状态
- ❌ 组件卸载时的竞态条件
- ❌ 没有类型安全
- ❌ 没有重试逻辑
- ❌ 没有缓存

**更好（审查和改进后）：**
```typescript
interface DataType {
  id: string;
  name: string;
  // ... 其他字段
}

interface UseDataResult {
  data: DataType[];
  loading: boolean;
  error: Error | null;
  refetch: () => void;
}

function useData(): UseDataResult {
  const [data, setData] = useState<DataType[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);
  const [refetchKey, setRefetchKey] = useState(0);

  useEffect(() => {
    let cancelled = false;
    const controller = new AbortController();

    async function fetchData() {
      try {
        setLoading(true);

        const response = await fetch('/api/data', {
          signal: controller.signal,
          headers: {
            'Content-Type': 'application/json',
          },
        });

        if (!response.ok) {
          throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }

        const json = await response.json();

        if (!cancelled) {
          setData(json);
          setError(null);
        }
      } catch (err) {
        if (err.name === 'AbortError') return;

        if (!cancelled) {
          setError(err as Error);
          console.error('获取数据失败:', err);
        }
      } finally {
        if (!cancelled) {
          setLoading(false);
        }
      }
    }

    fetchData();

    return () => {
      cancelled = true;
      controller.abort();
    };
  }, [refetchKey]);

  const refetch = () => setRefetchKey(k => k + 1);

  return { data, loading, error, refetch };
}
```

### 陷阱 3：忽视性能

**示例：**
```python
# ❌ AI 的第一次尝试（有效但慢 - O(n²)）
def find_duplicates(arr):
    duplicates = []
    for i in range(len(arr)):
        for j in range(i + 1, len(arr)):
            if arr[i] == arr[j] and arr[i] not in duplicates:
                duplicates.append(arr[i])
    return duplicates

# 总是问：
"时间复杂度是多少？我们能优化这个吗？"

# ✅ 优化版本（O(n)）
def find_duplicates(arr):
    """在数组中查找重复元素 - O(n) 时间, O(n) 空间"""
    seen = set()
    duplicates = set()

    for item in arr:
        if item in seen:
            duplicates.add(item)
        seen.add(item)

    return list(duplicates)

# 对于大型数据集，要求：
"展示处理 100 万个项目时最高效的算法"
```

### 陷阱 4：不测试边界情况

```python
# AI 生成这个：
def divide(a, b):
    return a / b

# 你必须测试：
- divide(10, 2)  # 正常情况
- divide(10, 0)  # 除以零 ❌
- divide(10, -2) # 负数
- divide(0, 10)  # 零分子
- divide(None, 2) # None 值 ❌
- divide("10", "2") # 错误类型 ❌
- divide(float('inf'), 2) # 无穷大
- divide(10**100, 10**-100) # 溢出

# 更好的实现：
def divide(a: float, b: float) -> float:
    """
    安全地除两个数字

    Args:
        a: 分子
        b: 分母

    Returns:
        除法的结果

    Raises:
        TypeError: 如果输入不是数字
        ValueError: 如果分母为零
    """
    if not isinstance(a, (int, float)) or not isinstance(b, (int, float)):
        raise TypeError("两个参数都必须是数字")

    if b == 0:
        raise ValueError("不能除以零")

    return a / b
```

## 📊 成本管理

### 跟踪你的支出

```python
# cost-tracker.py
from datetime import datetime
from typing import Dict, List
import json

class AIUsageTracker:
    """跨多个模型跟踪 AI API 使用和成本"""

    def __init__(self, log_file='ai_usage.json'):
        self.log_file = log_file
        self.usage = {
            'gpt-4': {'tokens': 0, 'cost': 0, 'requests': 0},
            'gpt-4o': {'tokens': 0, 'cost': 0, 'requests': 0},
            'gpt-3.5-turbo': {'tokens': 0, 'cost': 0, 'requests': 0},
            'claude-3-opus': {'tokens': 0, 'cost': 0, 'requests': 0},
            'claude-3-sonnet': {'tokens': 0, 'cost': 0, 'requests': 0},
        }

        # 每百万 token 的价格（截至 2025 年）
        self.prices = {
            'gpt-4': {'input': 30, 'output': 60},
            'gpt-4o': {'input': 5, 'output': 15},
            'gpt-3.5-turbo': {'input': 0.5, 'output': 1.5},
            'claude-3-opus': {'input': 15, 'output': 75},
            'claude-3-sonnet': {'input': 3, 'output': 15},
        }

    def track_request(self, model: str, input_tokens: int, output_tokens: int):
        """跟踪单个 API 请求"""
        if model not in self.prices:
            raise ValueError(f"未知模型: {model}")

        input_cost = input_tokens * self.prices[model]['input'] / 1_000_000
        output_cost = output_tokens * self.prices[model]['output'] / 1_000_000
        total_cost = input_cost + output_cost

        self.usage[model]['tokens'] += input_tokens + output_tokens
        self.usage[model]['cost'] += total_cost
        self.usage[model]['requests'] += 1

        # 记录到文件
        self._log_request(model, input_tokens, output_tokens, total_cost)

        return total_cost

    def _log_request(self, model, input_tokens, output_tokens, cost):
        """将请求追加到日志文件"""
        entry = {
            'timestamp': datetime.now().isoformat(),
            'model': model,
            'input_tokens': input_tokens,
            'output_tokens': output_tokens,
            'cost': cost
        }

        try:
            with open(self.log_file, 'a') as f:
                f.write(json.dumps(entry) + '\n')
        except Exception as e:
            print(f"记录请求失败: {e}")

    def report(self, detailed=False):
        """生成使用报告"""
        total_cost = sum(m['cost'] for m in self.usage.values())
        total_tokens = sum(m['tokens'] for m in self.usage.values())
        total_requests = sum(m['requests'] for m in self.usage.values())

        print(f"\n{'='*50}")
        print(f"💰 AI 使用报告 - {datetime.now().strftime('%Y-%m-%d %H:%M')}")
        print(f"{'='*50}\n")

        print(f"总支出: ${total_cost:.2f}")
        print(f"总 Token: {total_tokens:,}")
        print(f"总请求: {total_requests:,}")
        print(f"每次请求平均: ${total_cost/total_requests:.4f}\n" if total_requests > 0 else "\n")

        for model, data in sorted(self.usage.items(), key=lambda x: x[1]['cost'], reverse=True):
            if data['requests'] > 0:
                print(f"{model}:")
                print(f"  请求: {data['requests']:,}")
                print(f"  Token: {data['tokens']:,}")
                print(f"  成本: ${data['cost']:.2f}")
                print(f"  平均/请求: ${data['cost']/data['requests']:.4f}\n")

    def set_budget_alert(self, daily_limit: float):
        """如果每日支出超过限制则警报"""
        today_cost = self._get_today_cost()
        if today_cost >= daily_limit:
            print(f"⚠️  预算警报: 每日支出 (${today_cost:.2f}) 超过限制 (${daily_limit:.2f})")
            return True
        return False

    def _get_today_cost(self):
        """从日志计算今天的支出"""
        today = datetime.now().date()
        total = 0

        try:
            with open(self.log_file, 'r') as f:
                for line in f:
                    entry = json.loads(line)
                    entry_date = datetime.fromisoformat(entry['timestamp']).date()
                    if entry_date == today:
                        total += entry['cost']
        except FileNotFoundError:
            pass

        return total

# 使用示例
tracker = AIUsageTracker()

# 每次 API 调用后：
cost = tracker.track_request('gpt-4o', input_tokens=1000, output_tokens=500)
print(f"本次请求: ${cost:.4f}")

# 检查预算
tracker.set_budget_alert(daily_limit=10.00)

# 每日/每周报告：
tracker.report()
```

```python
def select_model(task_type, complexity, budget):
    """
    为任务选择正确的模型

    任务类型：
    - boilerplate: 简单、重复的代码
    - logic: 复杂算法、业务逻辑
```

function useCustomHook<T>(params: Params): Result<T> {
  const [data, setData] = useState<T | null>(null);

  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    // 带清理的异步逻辑

  }, [dependencies]);

  return { data, loading, error };
}

```python
# scripts/ai-review.py
import os
import sys
from openai import OpenAI

client = OpenAI(api_key=os.getenv('OPENAI_API_KEY'))

def review_code(file_path):
    """AI 审查单个文件"""
    with open(file_path, 'r') as f:
        code = f.read()

    prompt = f"""
    审查此代码：
    1. Bug 和逻辑错误
    2. 安全漏洞
    3. 性能问题
    4. 代码风格
    5. 缺少的边界情况

    文件: {file_path}

    ```
    {code}
    ```

    提供具体、可操作的反馈。
    """

    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}]
    )

    return response.choices[0].message.content

def main():
    files = sys.argv[2].split()  # --files file1.py file2.py

    reviews = []
    for file_path in files:
        if file_path.endswith(('.py', '.js', '.ts', '.tsx', '.java')):
            print(f"审查 {file_path}...")
            review = review_code(file_path)
            reviews.append(f"### {file_path}\n\n{review}\n")

    # 将审查写入文件以供 PR 评论
    with open('ai-review.md', 'w') as f:
        f.write('\n'.join(reviews))

if __name__ == '__main__':
    main()
```
