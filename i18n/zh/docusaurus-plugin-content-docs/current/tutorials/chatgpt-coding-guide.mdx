```python
from flask import jsonify, request
from werkzeug.security import check_password_hash
from models import User
import jwt
import datetime
from typing import Tuple, Dict
import re

def validate_email(email: str) -> bool:
    """Validate email format using regex."""
    pattern = r'^[\w\.-]+@[\w\.-]+\.\w+$'
    return re.match(pattern, email) is not None

def login_user(email: str, password: str) -> Tuple[Dict, int]:
    """
    Authenticate user and return JWT token.

    Args:
        email: User's email address
        password: Plain text password

    Returns:
        Tuple of (response_dict, http_status_code)

    Example:
        >>> login_user('user@example.com', 'password123')
        ({'token': 'eyJ0eXAiOiJKV1...'}, 200)

    Raises:
        ValueError: If email format is invalid
    """

    # Input validation
    if not email or not password:
        return {'error': 'Email and password are required'}, 400

    if not validate_email(email):
        return {'error': 'Invalid email format'}, 400

    # Find user
    user = User.query.filter_by(email=email).first()

    if not user:
        return {'error': 'Invalid credentials'}, 401

    # Verify password
    if not check_password_hash(user.password_hash, password):
        return {'error': 'Invalid credentials'}, 401

    # Generate JWT token
    token = jwt.encode({
        'user_id': user.id,
        'email': user.email,
        'exp': datetime.datetime.utcnow() + datetime.timedelta(hours=24)
    }, app.config['SECRET_KEY'], algorithm='HS256')

    return {
        'token': token,
        'user': {
            'id': user.id,
            'email': user.email,
            'name': user.name
        }
    }, 200
```

### 真实案例：React 无限渲染

**你的代码：**
```jsx
function UserProfile({ userId }) {
  const [user, setUser] = useState(null);

  useEffect(() => {
    fetch(`/api/users/${userId}`)
      .then(res => res.json())
      .then(setUser);
  }); // Bug: missing dependency array

  return <div>{user?.name}</div>;
}
```

**错误：**
```
Warning: Maximum update depth exceeded. This can happen when a component
calls setState inside useEffect without a dependency array.
```

**向 ChatGPT 提问：**
```
I'm getting "Maximum update depth exceeded" error in React.

Code:
[粘贴上面的代码]

Context:
- React 18 + TypeScript
- Trying to fetch user data on component mount
- Error appears immediately on page load

Please explain and fix.
```

**ChatGPT 回复：**
```
The problem is the missing dependency array in useEffect.

Current code:
useEffect(() => {
  fetch(`/api/users/${userId}`)
    .then(res => res.json())
    .then(setUser);
}); // Runs after EVERY render

What happens:
1. Component renders
2. useEffect runs, fetches data
3. setUser() causes re-render
4. useEffect runs again (because no dependency array)
5. Infinite loop!

Fix:
useEffect(() => {
  fetch(`/api/users/${userId}`)
    .then(res => res.json())
    .then(setUser);
}, [userId]); // Only run when userId changes

Complete improved version:
useEffect(() => {
  let cancelled = false;

  const fetchUser = async () => {
    try {
      const res = await fetch(`/api/users/${userId}`);
      if (!res.ok) throw new Error('Failed to fetch user');

      const data = await res.json();
      if (!cancelled) setUser(data);
    } catch (error) {
      console.error('Error fetching user:', error);
      // Handle error state
    }
  };

  fetchUser();

  return () => {
    cancelled = true; // Cleanup to prevent state update on unmounted component
  };
}, [userId]);
```

```python
def calculate_discount():
    data = request.get_json()

    price = data.get('price')
    user_tier = data.get('user_tier')
    promo_code = data.get('promo_code')

    if not price or price <= 0:
        return jsonify({'error': 'Invalid price'}), 400

    discount = 0

    # Tier discounts
    if user_tier == 'bronze': discount += 5
    elif user_tier == 'silver': discount += 10
    elif user_tier == 'gold': discount += 15

    # Promo code
    if promo_code == 'SAVE20':

    final_price = price * (1 - discount / 100)

    return jsonify({
        'original_price': price,
        'discount_percent': discount,
        'final_price': final_price
    })
```

## 专业提示

### 1. 始终验证 AI 代码

```python
# 不要盲目信任！
# ❌ 错误：不阅读就复制粘贴
# ✅ 正确：阅读、理解、测试

ai_generated_code = get_from_chatgpt()

# 阅读它
understand(ai_generated_code)

# 测试它
run_tests(ai_generated_code)

# 审查安全性
check_security(ai_generated_code)

# 然后使用它
```

### 2. 使用 ChatGPT 学习，而不仅仅是获取答案

```
# ❌ 错误："Fix my code"
# ✅ 正确："Explain why this doesn't work and teach me the concept"
```

### 3. 结合文档使用

```
# 最佳工作流程：
1. 阅读官方文档（15 分钟）
2. 向 ChatGPT 寻求澄清（5 分钟）
3. 尝试实现（20 分钟）
4. 如需要，使用 ChatGPT 调试（10 分钟）

总计：50 分钟学会新概念
```

### 4. 跟踪成本（GPT-4）

```python
# 如果使用 API
import openai

def count_tokens(text):
    # 近似：1 个 token ≈ 4 个字符
    return len(text) // 4

prompt = "Your prompt here"
tokens = count_tokens(prompt)
cost = tokens * $0.03 / 1000  # GPT-4 定价

print(f"Estimated cost: ${cost:.4f}")
```

## ⚠️ 常见陷阱

### 1. 幻觉函数

ChatGPT 可能会发明不存在的函数：

```python
# ChatGPT 可能会建议：
result = pandas.magic_function(df)  # ❌ 不存在！

# 始终在文档中验证
```

### 2. 过时的信息

训练数据截止日期（GPT-4 为 2024 年 1 月）：

```
# ❌ ChatGPT 不知道：
- 2024 年后发布的库
- 最新的 API 更改
- 最近的安全漏洞

# ✅ 解决方案：在提示词中提供文档
```

### 3. 过度工程化

ChatGPT 倾向于建议复杂的解决方案：

```python
# 你问："Store user preferences"

# ChatGPT 建议：Redis + RabbitMQ + Kafka + Microservices

# 现实：JSON 文件对你的用例来说已经足够了

# 始终质疑："这种复杂性有必要吗？"
```
